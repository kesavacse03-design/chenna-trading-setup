// Local instrument map (optional) — maps trading_symbol to instrument_key for equities
const instrumentMaps = { /** @type {Record<string, Map<string,string>>} */ };
function getInstrumentsPath(exchangePrefix) {
  // Expect files at workspaceRoot/.data/NSE.json/NSE.json or BSE.json/BSE.json
  // __dirname => .../chenna-CTS/backend, go up two => workspace root
  const root = path.resolve(__dirname, '..', '..');
  const base = exchangePrefix.startsWith('BSE') ? 'BSE.json' : 'NSE.json';
  return path.join(root, '.data', base, base.replace('.json', '') + '.json');
}
function loadInstrumentMap(exchangePrefix) {
  if (instrumentMaps[exchangePrefix]) return instrumentMaps[exchangePrefix];
  const p = getInstrumentsPath(exchangePrefix);
  try {
    if (!fs.existsSync(p)) return null;
    const raw = fs.readFileSync(p, 'utf8');
    const arr = JSON.parse(raw);
    const map = new Map();
    for (const it of arr) {
      if (!it) continue;
      if (it.segment !== exchangePrefix) continue;
      // Focus on common cash types
      if (!['EQ','BE','ST','SM','N0','SME'].includes(it.instrument_type)) continue;
      const sym = String(it.trading_symbol || '').toUpperCase();
      const key = String(it.instrument_key || it.asset_key || '').toUpperCase();
      if (sym && key && !map.has(sym)) map.set(sym, key);
    }
    instrumentMaps[exchangePrefix] = map;
    if (VERBOSE) console.log('[CTS] Loaded instruments', exchangePrefix, 'count=', map.size);
    return map;
  } catch (e) {
    if (VERBOSE) console.warn('[CTS] Failed to load instruments for', exchangePrefix, 'at', p);
    return null;
  }
}
function resolveInstrumentKey(sym, exchangePrefix) {
  const map = loadInstrumentMap(exchangePrefix);
  const upper = String(sym).toUpperCase();
  if (map && map.has(upper)) return map.get(upper);
  return null;
}

// Simple filesystem cache for OHLCV requests
const CACHE_DIR = path.join(__dirname, 'cache', 'ohlcv');
function ensureCacheDir() { try { fs.mkdirSync(CACHE_DIR, { recursive: true }); } catch {} }
function cachePathFor(key, from, to) {
  const safe = String(key).replace(/[^A-Z0-9_|:-]/gi, '_');
  return path.join(CACHE_DIR, `${safe}_${from}_${to}.json`);
}

// Helper to extract price/timestamp from various Upstox response shapes
function extractPriceAndTime(node) {
  if (!node || typeof node !== 'object') return { price: NaN, ts: null, field: null };
  const candidates = [
    'ltp',
    'last_price',
    'lastPrice',
    'last_traded_price',
    'lastTradedPrice',
    'close',
  ];
  let price = NaN;
  let field = null;
  for (const f of candidates) {
    const v = node[f];
    const n = Number(v);
    if (!Number.isNaN(n)) { price = n; field = f; break; }
  }
  const ts = node.timestamp || node.updated_at || node.exchange_timestamp || null;
  return { price, ts, field };
}
